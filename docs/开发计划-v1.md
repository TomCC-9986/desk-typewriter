# 桌面打字机 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** An Electron desktop app (Mac + Windows) where friends type notes on a bottom-of-screen typewriter and the note prints, line by line, on the recipient's screen from the top edge; relayed by a Cloudflare Worker with one Durable Object inbox per user.

**Architecture:** `server/` is a Cloudflare Worker (`/register`, `/ws`) plus an `Inbox` Durable Object (SQLite-backed, WebSocket Hibernation) that holds friends, pending requests and an offline queue; DO→DO calls use RPC. `app/` is Electron: the main process owns the WebSocket (`Connection`), a JSON `Store`, and five transparent windows (typewriter, feeder, note, fx, panel). Pure logic (layout, typewriter model, connection, trash target, file naming) lives in `app/src/lib` and is unit-tested; protocol types live in top-level `shared/`.

**Tech Stack:** TypeScript 5, Cloudflare Workers + Durable Objects + KV, wrangler 4, vitest 3.2 + @cloudflare/vitest-pool-workers 0.8, Electron 38, esbuild, electron-builder 26. No UI framework, no audio assets (WebAudio-synthesized SFX).

```ts
export interface Note { id: string; from: string; fromName: string; text: string; sentAt: number }

export type ClientMsg =
  | { type: "note"; id: string; to: string; text: string; sentAt: number }
  | { type: "ack"; id: string }
  | { type: "friend_request"; code: string }

export type ServerMsg =
  | { type: "hello"; me: Friend; code: string; friends: Friend[]; pending: Friend[] }

  | { type: "friend_request"; from: string; nickname: string }

  | { type: "error"; reason: string };
```

```ts
export const CODE_ALPHABET = "ABCDEFGHJKMNPQRSTUVWXYZ23456789";

const cryptoRand = (n: number) => crypto.getRandomValues(new Uint32Array(1))[0] % n;

export function generateCode(rand: (n: number) => number = cryptoRand): string {
  let s = "";
  for (let i = 0; i < 6; i++) s += CODE_ALPHABET[rand(CODE_ALPHABET.length) % CODE_ALPHABET.length];
  return `${s.slice(0, 3)}-${s.slice(3)}`;
}

export function normalizeCode(input: string): string | null {
  const s = input.toUpperCase().replace(/[^A-Z0-9]/g, "");
  if (s.length !== 6 || [...s].some((c) => !CODE_ALPHABET.includes(c))) return null;
  return `${s.slice(0, 3)}-${s.slice(3)}`;
}
```

```ts
import { SELF, env, runInDurableObject } from "cloudflare:test";
import type { ClientMsg, ServerMsg } from "../../shared/protocol";

export interface Registered { userId: string; secret: string; code: string; nickname: string }

export async function register(nickname: string): Promise<Registered> {
  const res = await SELF.fetch("http://x/register", { method: "POST", body: JSON.stringify({ nickname }) });
  if (!res.ok) throw new Error(`register failed ${res.status}`);
  return res.json();
}

export interface Client {
  next<T extends ServerMsg["type"]>(type?: T): Promise<Extract<ServerMsg, { type: T }>>;
  drain(ms: number): Promise<ServerMsg[]>;
  send(m: ClientMsg): void;
  close(): void;
}

export async function connect(u: { userId: string; secret: string }): Promise<Client> {
  const res = await SELF.fetch(`http://x/ws?uid=${u.userId}&secret=${u.secret}`, { headers: { Upgrade: "websocket" } });
  const ws = res.webSocket;
  if (!ws) throw new Error(`no websocket (status ${res.status})`);
  ws.accept();
  const inbox: ServerMsg[] = [];
  const waiters: Array<() => void> = [];
  ws.addEventListener("message", (e) => {
    inbox.push(JSON.parse(e.data as string));
    waiters.splice(0).forEach((w) => w());
  });
  return {
    async next(type) {
      for (;;) {
        const i = inbox.findIndex((m) => !type || m.type === type);
        if (i >= 0) return inbox.splice(i, 1)[0] as never;
        await new Promise<void>((resolve, reject) => {
          const t = setTimeout(() => reject(new Error(`timeout waiting for ${type ?? "any"}`)), 2000);
          waiters.push(() => { clearTimeout(t); resolve(); });
        });
      }
    },
    async drain(ms) {
      await new Promise((r) => setTimeout(r, ms));
      return inbox.splice(0);
    },
    send: (m) => ws.send(JSON.stringify(m)),
    close: () => ws.close(1000, "test"),
  };
}

export function inboxStub(userId: string) {
  return env.INBOX.get(env.INBOX.idFromName(userId));
}

export async function eventually(check: () => Promise<boolean>, ms = 2000): Promise<void> {
  const end = Date.now() + ms;
  while (Date.now() < end) {
    if (await check()) return;
    await new Promise((r) => setTimeout(r, 20));
  }
  throw new Error("condition not met in time");
}

export async function waitOffline(userId: string): Promise<void> {
  await eventually(() =>
    runInDurableObject(inboxStub(userId), async (_i, state) =>
      state.getWebSockets().every((ws) => ws.readyState !== 1),
    ),
  );
}

/** Registers two users, connects both, makes them friends, consumes setup messages. */
export async function befriend(nameA = "小王", nameB = "阿杰") {
  const a = await register(nameA);
  const b = await register(nameB);
  const ca = await connect(a);
  const cb = await connect(b);
  await ca.next("hello");
  await cb.next("hello");
  ca.send({ type: "friend_request", code: b.code });
  await ca.next("request_sent");
  await cb.next("friend_request");
  cb.send({ type: "friend_response", from: a.userId, accepted: true });
  await cb.next("friend_added");
  await ca.next("friend_added");
  return { a, b, ca, cb };
}
```

```ts
import { generateCode } from "../../shared/code";
import { Inbox } from "./inbox";

export { Inbox };

export interface Env {
  USERS: KVNamespace;
  INBOX: DurableObjectNamespace<Inbox>;
}

interface UserRecord { secretHash: string; code: string; nickname: string }

const json = (body: unknown, status = 200) =>
  new Response(JSON.stringify(body), { status, headers: { "content-type": "application/json" } });

const hex = (bytes: Uint8Array) => [...bytes].map((b) => b.toString(16).padStart(2, "0")).join("");

async function sha256(s: string): Promise<string> {
  return hex(new Uint8Array(await crypto.subtle.digest("SHA-256", new TextEncoder().encode(s))));
}

async function register(req: Request, env: Env): Promise<Response> {
  let body: { nickname?: unknown };
  try { body = await req.json(); } catch { return json({ error: "bad_json" }, 400); }
  const nickname = typeof body.nickname === "string" ? [...body.nickname.trim()].slice(0, 16).join("") : "";
  if (!nickname) return json({ error: "nickname_required" }, 400);

  let code = generateCode();
  for (let i = 0; await env.USERS.get(`code:${code}`); i++) {
    if (i > 10) return json({ error: "code_exhausted" }, 500);
    code = generateCode();
  }
  const userId = crypto.randomUUID();
  const secret = hex(crypto.getRandomValues(new Uint8Array(32)));
  const record: UserRecord = { secretHash: await sha256(secret), code, nickname };
  await env.USERS.put(`code:${code}`, userId);
  await env.USERS.put(`user:${userId}`, JSON.stringify(record));
  await env.INBOX.get(env.INBOX.idFromName(userId)).init({ userId, nickname }, code);
  return json({ userId, secret, code, nickname });
}

async function connect(req: Request, env: Env, url: URL): Promise<Response> {
  if (req.headers.get("Upgrade") !== "websocket") return new Response("expected websocket", { status: 426 });
  const uid = url.searchParams.get("uid") ?? "";
  const secret = url.searchParams.get("secret") ?? "";
  const raw = await env.USERS.get(`user:${uid}`);
  if (!raw) return new Response("unauthorized", { status: 401 });
  const rec = JSON.parse(raw) as UserRecord;
  if (rec.secretHash !== (await sha256(secret))) return new Response("unauthorized", { status: 401 });
  return env.INBOX.get(env.INBOX.idFromName(uid)).fetch(req);
}

export default {
  async fetch(req: Request, env: Env): Promise<Response> {
    const url = new URL(req.url);
    if (url.pathname === "/register" && req.method === "POST") return register(req, env);
    if (url.pathname === "/ws") return connect(req, env, url);
    if (url.pathname === "/health") return new Response("ok");
    return new Response("not found", { status: 404 });
  },
} satisfies ExportedHandler<Env>;
```

```ts
import { DurableObject } from "cloudflare:workers";
import type { Friend, ServerMsg } from "../../shared/protocol";
import type { Env } from "./index";

export type FriendMap = Record<string, string>; // userId -> nickname

export const toList = (m: FriendMap): Friend[] =>
  Object.entries(m).map(([userId, nickname]) => ({ userId, nickname }));

export class Inbox extends DurableObject<Env> {
  async init(me: Friend, code: string): Promise<void> {
    await this.ctx.storage.put({ me, code });
  }

  protected async me(): Promise<Friend> {
    return (await this.ctx.storage.get<Friend>("me"))!;
  }
  protected async friends(): Promise<FriendMap> {
    return (await this.ctx.storage.get<FriendMap>("friends")) ?? {};
  }
  protected async pending(): Promise<FriendMap> {
    return (await this.ctx.storage.get<FriendMap>("pending")) ?? {};
  }
  protected reply(ws: WebSocket, msg: ServerMsg): void {
    ws.send(JSON.stringify(msg));
  }
  protected broadcast(msg: ServerMsg): void {
    const s = JSON.stringify(msg);
    for (const ws of this.ctx.getWebSockets()) {
      try { ws.send(s); } catch { /* socket closing */ }
    }
  }
  protected online(): boolean {
    return this.ctx.getWebSockets().some((ws) => ws.readyState === 1);
  }

  async fetch(_req: Request): Promise<Response> {
    const pair = new WebSocketPair();
    const server = pair[1];
    this.ctx.acceptWebSocket(server);
    this.reply(server, {
      type: "hello",
      me: await this.me(),
      code: (await this.ctx.storage.get<string>("code")) ?? "",
      friends: toList(await this.friends()),
      pending: toList(await this.pending()),
    });
    return new Response(null, { status: 101, webSocket: pair[0] });
  }

  async webSocketClose(ws: WebSocket, code: number): Promise<void> {
    try { ws.close(code, "bye"); } catch { /* already closed */ }
  }
}
```

```ts
    const reject = (reason: string) => this.reply(ws, { type: "status", id: msg.id, state: "rejected", reason });
    if (typeof msg.text !== "string" || !msg.text.trim()) return reject("empty");

    if (this.sent.length >= RATE_PER_MIN) return reject("rate_limited");

      id: msg.id, from: me.userId, fromName: me.nickname, text: msg.text, sentAt: msg.sentAt,
    });
    this.reply(ws, state === "rejected"

    if (!this.online()) return "queued";
```

```ts
export function charWidth(ch: string): 1 | 2 {
  const c = ch.codePointAt(0) ?? 0;
  if (
    (c >= 0x1100 && c <= 0x115f) ||
    (c >= 0x2e80 && c <= 0xa4cf) ||
    (c >= 0xac00 && c <= 0xd7a3) ||
    (c >= 0xf900 && c <= 0xfaff) ||
    (c >= 0xfe30 && c <= 0xfe4f) ||
    (c >= 0xff00 && c <= 0xff60) ||
    (c >= 0xffe0 && c <= 0xffe6) ||
    (c >= 0x1f300 && c <= 0x1faff) ||
    (c >= 0x20000 && c <= 0x3fffd)
  ) return 2;
  return 1;
}

/** Splits items into lines of at most `cols` width units. "\n" items break the line and are dropped. */
export function wrapItems<T>(items: T[], ch: (t: T) => string, cols: number): T[][] {
  const lines: T[][] = [];
  let cur: T[] = [];
  let w = 0;
  for (const item of items) {
    const c = ch(item);
    if (c === "\n") { lines.push(cur); cur = []; w = 0; continue; }
    const cw = charWidth(c);
    if (w + cw > cols && cur.length) { lines.push(cur); cur = []; w = 0; }
    cur.push(item);
    w += cw;
  }
  lines.push(cur);
  return lines;
}

export function wrap(text: string, cols: number): string[] {
  return wrapItems([...text], (c) => c, cols).map((l) => l.join(""));
}
```

```ts
import { MAX_TEXT } from "@shared/protocol";

export interface Cell { ch: string; struck: boolean }

export class TypewriterModel {
  cells: Cell[] = [];

  constructor(initial = "") {
    this.type(initial);
  }

  get length(): number {
    return this.cells.filter((c) => !c.struck).length;
  }

  text(): string {
    return this.cells.filter((c) => !c.struck).map((c) => c.ch).join("");
  }

  /** Returns the characters actually accepted (truncated at MAX_TEXT). */
  type(s: string): string {
    let accepted = "";
    for (const ch of s) {
      if (this.length >= MAX_TEXT) break;
      this.cells.push({ ch, struck: false });
      accepted += ch;
    }
    return accepted;
  }

  /** Strikes the last live char; a newline is removed instead. Returns whether anything changed. */
  backspace(): boolean {
    for (let i = this.cells.length - 1; i >= 0; i--) {
      const c = this.cells[i];
      if (c.struck) continue;
      if (c.ch === "\n") this.cells.splice(i, 1);
      else c.struck = true;
      return true;
    }
    return false;
  }

  clear(): void {
    this.cells = [];
  }
}
```

```ts
import type { ClientMsg, ServerMsg } from "@shared/protocol";

export interface SocketLike {
  send(d: string): void;
  close(): void;
  onopen: (() => void) | null;
  onclose: (() => void) | null;
  onmessage: ((e: { data: unknown }) => void) | null;
  onerror: (() => void) | null;
}

export interface ConnectionOpts {
  url: () => string | null;
  onMessage: (m: ServerMsg) => void;
  onStatus: (online: boolean) => void;
  create?: (url: string) => SocketLike;
  setTimer?: (fn: () => void, ms: number) => unknown;
  clearTimer?: (t: unknown) => void;
}

export function backoffMs(attempt: number): number {
  return Math.min(30_000, 1000 * 2 ** attempt);
}

export class Connection {
  online = false;
  private sock: SocketLike | null = null;
  private attempt = 0;
  private timer: unknown = null;
  private stopped = true;

  constructor(private o: ConnectionOpts) {}

  start(): void {
    this.stopped = false;
    this.open();
  }

  stop(): void {
    this.stopped = true;
    if (this.timer !== null) (this.o.clearTimer ?? ((t) => clearTimeout(t as NodeJS.Timeout)))(this.timer);
    this.timer = null;
    const s = this.sock;
    this.sock = null;
    s?.close();
    this.setOnline(false);
  }

  restart(): void {
    this.stop();
    this.attempt = 0;
    this.start();
  }

  send(m: ClientMsg): boolean {
    if (!this.online || !this.sock) return false;
    this.sock.send(JSON.stringify(m));
    return true;
  }

  private open(): void {
    const url = this.o.url();
    if (!url || this.stopped) return;
    const create = this.o.create ?? ((u: string) => new WebSocket(u) as unknown as SocketLike);
    const s = create(url);
    this.sock = s;
    s.onopen = () => {
      if (this.sock !== s) return;
      this.attempt = 0;
      this.setOnline(true);
    };
    s.onmessage = (e) => {
      if (this.sock !== s) return;
      let msg: ServerMsg;
      try { msg = JSON.parse(String(e.data)); } catch { return; }
      this.o.onMessage(msg);
    };
    s.onerror = () => {};
    s.onclose = () => {
      if (this.sock !== s) return;
      this.sock = null;
      this.setOnline(false);
      this.schedule();
    };
  }

  private schedule(): void {
    if (this.stopped) return;
    const ms = backoffMs(this.attempt++);
    this.timer = (this.o.setTimer ?? setTimeout)(() => {
      this.timer = null;
      this.open();
    }, ms);
  }

  private setOnline(v: boolean): void {
    if (this.online === v) return;
    this.online = v;
    this.o.onStatus(v);
  }
}
```

```ts
export interface Point { x: number; y: number }
export interface Rect { x: number; y: number; width: number; height: number }
export interface DockInfo { tileSize: number; iconCount: number }

export const DEFAULT_DOCK: DockInfo = { tileSize: 48, iconCount: 14 };

/** Estimated screen position of the Trash (mac Dock) or Recycle Bin (Windows desktop, top-left). */
export function trashTarget(platform: string, bounds: Rect, work: Rect, dock: DockInfo = DEFAULT_DOCK): Point {
  if (platform !== "darwin") return { x: work.x + 40, y: work.y + 40 };
  const half = dock.tileSize / 2;
  const len = dock.iconCount * dock.tileSize + 24;
  const bottomGap = bounds.y + bounds.height - (work.y + work.height);
  const leftGap = work.x - bounds.x;
  const rightGap = bounds.x + bounds.width - (work.x + work.width);
  const alongX = Math.min(bounds.x + bounds.width - half, Math.round(bounds.x + bounds.width / 2 + len / 2 - half));
  const alongY = Math.min(bounds.y + bounds.height - half, Math.round(bounds.y + bounds.height / 2 + len / 2 - half));
  if (bottomGap > 8) return { x: alongX, y: Math.round(bounds.y + bounds.height - bottomGap / 2) };
  if (leftGap > 8) return { x: Math.round(bounds.x + leftGap / 2), y: alongY };
  if (rightGap > 8) return { x: Math.round(bounds.x + bounds.width - rightGap / 2), y: alongY };
  return { x: alongX, y: bounds.y + bounds.height - 10 }; // auto-hidden dock
}

export function countTiles(defaultsOutput: string): number {
  return (defaultsOutput.match(/"tile-data"/g) ?? []).length;
}
```

```ts
    for (const s of this.store.data.stuck) this.createNote(s);
  }

  // ---- fx overlay (crumpled ball flying to the trash) ----
  fly(from: Point, to: Point): Promise<void> {
    const d = screen.getDisplayNearestPoint(from);
    if (!alive(this.fx)) {
      this.fx = make({ ...d.bounds, focusable: false, enableLargerThanScreen: true });
      floatTop(this.fx);
      this.fx.setIgnoreMouseEvents(true);
      void this.fx.loadFile(page("fx"));
    } else {
      this.fx.setBounds(d.bounds);
    }
    const fx = this.fx;
    const id = crypto.randomUUID();
    return new Promise((resolve) => {
      this.flights.set(id, resolve);
      setTimeout(() => this.flightDone(id), 3000); // safety net
      void loaded(fx).then(() => {
        fx.showInactive();
        const req: FlyRequest = { id, from, to, origin: { x: d.bounds.x, y: d.bounds.y } };
        fx.webContents.send("fx:fly", req);
      });
    });
  }

  flightDone(id: string): void {
    const resolve = this.flights.get(id);
    if (!resolve) return;
    this.flights.delete(id);
    resolve();
    if (this.flights.size === 0) this.fx?.hide();
  }

  // ---- panel ----
  showPanel(section?: string): void {
    if (!alive(this.panel)) {
      this.panel = new BrowserWindow({
        width: 460, height: 680, title: "桌面打字机", show: false, resizable: false,
        webPreferences: { preload: join(__dirname, "preload.js"), contextIsolation: true, sandbox: true },
      });
      void this.panel.loadFile(page("panel"));
      this.panel.once("ready-to-show", () => this.panel?.show());
    } else {
      this.panel.show();
    }
    app.focus({ steal: true });
    this.panel.focus();
    if (section) {
      const p = this.panel;
      void loaded(p).then(() => p.webContents.send("panel:section", section));
    }
  }

  broadcast(channel: string, ...args: unknown[]): void {
    for (const w of BrowserWindow.getAllWindows()) if (!w.isDestroyed()) w.webContents.send(channel, ...args);
  }
```

```ts
import { app, screen } from "electron";
import { normalizeCode } from "@shared/code";
import type { Note, ServerMsg } from "@shared/protocol";
import { Connection } from "../lib/connection";
import { SeenIds } from "../lib/seen-ids";
import { trashTarget, type Point } from "../lib/trash-target";
import type { AppView, SendResult, TypewriterInit } from "../shared/ipc-types";
import type { Settings, Store } from "./store";
import { readDockInfo, trashNoteFile } from "./trash";
import type { Windows } from "./windows";

export const AUTOTRASH_MS = Number(process.env.DESK_AUTOTRASH_MS) || 30 * 60_000;

type FriendResult = { ok: boolean; reason?: string };

export class Hub {
  online = false;
  onTray: (() => void) | null = null;
  private conn: Connection;
  private seen: SeenIds;
  private shownRequests = new Set<string>();
  private statusWaiters = new Map<string, (r: SendResult) => void>();
  private friendWaiter: ((r: FriendResult) => void) | null = null;

  constructor(private store: Store, private win: Windows) {
    this.seen = new SeenIds(store.data.seen);
    this.conn = new Connection({
      url: () => this.wsUrl(),
      onMessage: (m) => this.onServer(m),
      onStatus: (v) => { this.online = v; this.changed(); },
    });
  }

  start(): void { this.conn.start(); }

  private changed(): void {
    this.win.broadcast("state:changed");
    this.onTray?.();
  }

  private wsUrl(): string | null {
    const id = this.store.data.identity;
    if (!id) return null;
    const base = this.store.data.settings.serverUrl.replace(/^http/, "ws").replace(/\/$/, "");
    return `${base}/ws?uid=${encodeURIComponent(id.userId)}&secret=${encodeURIComponent(id.secret)}`;
  }

  view(): AppView {
    const d = this.store.data;
    const id = d.identity;
    return {
      me: id ? { userId: id.userId, nickname: id.nickname, code: id.code } : null,
      friends: d.friends, pending: d.pending, history: d.history, settings: d.settings,
      online: this.online, autoTrashMs: AUTOTRASH_MS, heldCount: d.held.length,
    };
  }

  typewriterInit(): TypewriterInit {
    const d = this.store.data;
    return { friends: d.friends, lastTo: d.lastTo, draft: d.draft, volume: d.settings.volume, online: this.online };
  }

  // ---- server → app ----
  private onServer(m: ServerMsg): void {
    switch (m.type) {
      case "hello":
        this.store.update((d) => {
          d.friends = m.friends;
          d.pending = m.pending;
          if (d.identity) { d.identity.code = m.code; d.identity.nickname = m.me.nickname; }
        });
        for (const p of m.pending) this.showRequest(p.userId, p.nickname);
        this.changed();
        break;
      case "note":
        this.receive({ id: m.id, from: m.from, fromName: m.fromName, text: m.text, sentAt: m.sentAt });
        break;
      case "status":
        this.statusWaiters.get(m.id)?.({ state: m.state, reason: m.reason });
        this.statusWaiters.delete(m.id);
        break;
      case "request_sent":
        this.friendWaiter?.({ ok: true });
        break;
      case "error":
        this.friendWaiter?.({ ok: false, reason: m.reason });
        break;
      case "friend_request":
        this.store.update((d) => {
          if (!d.pending.some((p) => p.userId === m.from)) d.pending.push({ userId: m.from, nickname: m.nickname });
        });
        this.showRequest(m.from, m.nickname);
        this.changed();
        break;
      case "friend_added":
        this.store.update((d) => {
          d.friends = [...d.friends.filter((f) => f.userId !== m.userId), { userId: m.userId, nickname: m.nickname }];
          d.pending = d.pending.filter((p) => p.userId !== m.userId);
        });
        this.changed();
        break;
      case "friend_removed":
        this.store.update((d) => {
          d.friends = d.friends.filter((f) => f.userId !== m.userId);
          if (d.lastTo === m.userId) d.lastTo = null;
        });
        this.changed();
        break;
    }
  }

  private showRequest(from: string, nickname: string): void {
    if (this.shownRequests.has(from)) return;
    this.shownRequests.add(from);
    void this.win.print({ kind: "request", from, nickname });
  }

  private receive(note: Note): void {
    if (this.seen.add(note.id)) {
      const dnd = this.store.data.settings.dnd;
      this.store.update((d) => {
        d.seen = this.seen.toJSON();
        d.history = [{ ...note, receivedAt: Date.now() }, ...d.history].slice(0, 200);
        if (dnd) d.held.push(note);
      });
      if (!dnd) void this.win.print({ kind: "note", note });
      this.changed();
    }
    this.conn.send({ type: "ack", id: note.id }); // after persisting; duplicates are acked too
  }

  // ---- app → server ----
  sendNote(to: string, text: string): Promise<SendResult> {
    const id = crypto.randomUUID();
    if (!this.conn.send({ type: "note", id, to, text, sentAt: Date.now() })) return Promise.resolve({ state: "offline" });
    return new Promise((resolve) => {
      const t = setTimeout(() => { this.statusWaiters.delete(id); resolve({ state: "offline" }); }, 8000);
      this.statusWaiters.set(id, (r) => {
        clearTimeout(t);
        if (r.state !== "rejected") this.store.update((d) => { d.lastTo = to; d.draft = ""; });
        resolve(r);
      });
    });
  }

  addFriend(code: string): Promise<FriendResult> {
    const norm = normalizeCode(code);
    if (!norm) return Promise.resolve({ ok: false, reason: "bad_code" });
    if (!this.conn.send({ type: "friend_request", code: norm })) return Promise.resolve({ ok: false, reason: "offline" });
    return new Promise((resolve) => {
      const t = setTimeout(() => { this.friendWaiter = null; resolve({ ok: false, reason: "timeout" }); }, 6000);
      this.friendWaiter = (r) => { clearTimeout(t); this.friendWaiter = null; resolve(r); };
    });
  }

  respond(from: string, accepted: boolean): void {
    this.conn.send({ type: "friend_response", from, accepted });
    this.shownRequests.delete(from);
    this.store.update((d) => { d.pending = d.pending.filter((p) => p.userId !== from); });
    this.changed();
  }

  removeFriend(userId: string): void { this.conn.send({ type: "remove_friend", userId }); }
  blockFriend(userId: string): void { this.conn.send({ type: "block", userId }); }

  async register(nickname: string, serverUrl: string): Promise<FriendResult> {
    const base = serverUrl.trim().replace(/\/$/, "");
    try {
      const res = await fetch(`${base}/register`, {
        method: "POST", headers: { "content-type": "application/json" }, body: JSON.stringify({ nickname }),
      });
      const body = (await res.json()) as { userId: string; secret: string; code: string; nickname: string; error?: string };
      if (!res.ok) return { ok: false, reason: body.error ?? `http_${res.status}` };
      this.store.update((d) => {
        d.identity = { userId: body.userId, secret: body.secret, code: body.code, nickname: body.nickname };
        d.settings.serverUrl = base;
      });
      this.conn.restart();
      this.changed();
      return { ok: true };
    } catch {
      return { ok: false, reason: "network" };
    }
  }

  setSettings(patch: Partial<Settings>): void {
    const before = { ...this.store.data.settings };
    this.store.update((d) => { d.settings = { ...d.settings, ...patch }; });
    const after = this.store.data.settings;
    if (after.serverUrl !== before.serverUrl) this.conn.restart();
    if (after.openAtLogin !== before.openAtLogin) app.setLoginItemSettings({ openAtLogin: after.openAtLogin });
    if (after.feedPosition !== before.feedPosition) this.win.repositionFeeder();
    if (before.dnd && !after.dnd) {
      const held = this.store.data.held;
      this.store.update((d) => { d.held = []; });
      for (const note of held) void this.win.print({ kind: "note", note });
    }
    this.changed();
  }

  async autoTrash(note: Note, from: Point): Promise<void> {
    const d = screen.getDisplayNearestPoint(from);
    const to = trashTarget(process.platform, d.bounds, d.workArea, await readDockInfo());
    await this.win.fly(from, to);
    await trashNoteFile(note);
  }
}
```

```ts
/** 32×32 BGRA page icon. Mac: outline only (template). Windows: white page with dark outline. */
export function trayBitmap(filled: boolean): Buffer {
  const S = 32;
  const buf = Buffer.alloc(S * S * 4);
  const set = (x: number, y: number, v: number) => {
    const i = (y * S + x) * 4;
    buf[i] = v; buf[i + 1] = v; buf[i + 2] = v; buf[i + 3] = 255;
  };
  for (let y = 3; y <= 28; y++) {
    for (let x = 7; x <= 24; x++) {
      const border = x <= 8 || x >= 23 || y <= 4 || y >= 27;
      const line = (y === 11 || y === 12 || y === 16 || y === 17 || y === 21 || y === 22) && x >= 11 && x <= 20;
      if (border || line) set(x, y, 20);
      else if (filled) set(x, y, 250);
    }
  }
  return buf;
}
```

```ts
import type { Note } from "@shared/protocol";
import { charWidth, wrap } from "../lib/layout";

export const COLS = 24;     // width units per line (12 CJK chars)
export const UNIT_EM = 0.55; // one width unit in em
export const LINE_H = 28;
export const PAD = 36;       // top padding + bottom margin inside the paper

const rnd = (i: number, salt: number) => {
  const x = Math.sin(i * 12.9898 + salt * 78.233) * 43758.5453;
  return x - Math.floor(x);
};

/** One typed character with deterministic "real typewriter" wobble and ink variance. */
export function charSpan(ch: string, i: number, struck = false): HTMLSpanElement {
  const s = document.createElement("span");
  s.className = struck ? "ch struck" : "ch";
  s.textContent = ch;
  s.style.width = `${charWidth(ch) * UNIT_EM}em`;
  s.style.setProperty("--r", `${((rnd(i, 1) - 0.5) * 3).toFixed(2)}deg`);
  s.style.setProperty("--y", `${((rnd(i, 2) - 0.5) * 1.2).toFixed(2)}px`);
  s.style.setProperty("--ink", (0.72 + rnd(i, 3) * 0.28).toFixed(2));
  return s;
}

export function signature(n: Note): string {
  const d = new Date(n.sentAt);
  return `— ${n.fromName} ${String(d.getHours()).padStart(2, "0")}:${String(d.getMinutes()).padStart(2, "0")}`;
}

/** Fully printed paper (used by stuck notes). Returns the number of rows including the signature. */
export function renderStatic(lines: HTMLElement, note: Note): number {
  let i = 0;
  const rows = wrap(note.text, COLS);
  for (const r of rows) {
    const row = document.createElement("div");
    row.className = "line";
    for (const ch of r) row.append(charSpan(ch, i++));
    lines.append(row);
  }
  const sig = document.createElement("div");
  sig.className = "line sig";
  sig.textContent = signature(note);
  lines.append(sig);
  return rows.length + 1;
}

export const paperHeight = (rows: number, extra = 0) => rows * LINE_H + PAD + extra;
export const wait = (ms: number) => new Promise<void>((r) => setTimeout(r, ms));
```

```ts
let ctx: AudioContext | null = null;
let volume = 0.6;

export function setVolume(v: number): void { volume = v; }

const ac = () => (ctx ??= new AudioContext());

function noise(dur: number, freq: number, q: number, gain: number): void {
  if (volume <= 0) return;
  const c = ac();
  const len = Math.max(1, Math.floor(c.sampleRate * dur));
  const buf = c.createBuffer(1, len, c.sampleRate);
  const data = buf.getChannelData(0);
  for (let i = 0; i < len; i++) data[i] = (Math.random() * 2 - 1) * (1 - i / len) ** 4;
  const src = c.createBufferSource();
  src.buffer = buf;
  const f = c.createBiquadFilter();
  f.type = "bandpass"; f.frequency.value = freq; f.Q.value = q;
  const g = c.createGain();
  g.gain.value = gain * volume;
  src.connect(f).connect(g).connect(c.destination);
  src.start();
}

function tone(freq: number, dur: number, gain: number): void {
  if (volume <= 0) return;
  const c = ac();
  const o = c.createOscillator();
  o.type = "sine"; o.frequency.value = freq;
  const g = c.createGain();
  const t = c.currentTime;
  g.gain.setValueAtTime(gain * volume, t);
  g.gain.exponentialRampToValueAtTime(0.0001, t + dur);
  o.connect(g).connect(c.destination);
  o.start(t); o.stop(t + dur);
}
```

```ts
function enqueue(text: string): void {
  pendingChars.push(...text);
  void pump();

async function pump(): Promise<void> {
  if (pumping) return;
  pumping = true;
  while (pendingChars.length) {
    const ch = pendingChars.shift()!;

  }
  pumping = false;
}

  if (composing) return;
```

```ts
const MAX_SLOTS = 5;
const hung = new Set<HTMLElement>();
const queue: PrintItem[] = [];

let printing = false;
let dragging = false;

const jitter = (a: number, b: number) => a + Math.random() * (b - a);

async function refresh(): Promise<void> {
  view = await window.desk.invoke<AppView>("state:get");
  setVolume(view.settings.volume);

function slotX(slot: number): number {
  const W = window.innerWidth;
  const pos = view?.settings.feedPosition ?? "center";
  if (pos === "left") return 20 + slot * SLOT_W;
  if (pos === "right") return W - 300 - slot * SLOT_W;

}

async function pump(): Promise<void> {
  if (printing) return;
  printing = true;
  if (!view) await refresh();

function hang(paper: HTMLElement, item: PrintItem): void {
  hung.add(paper);
  paper.addEventListener("pointerenter", () => window.desk.send("feeder:mouse", false));
  paper.addEventListener("pointerleave", () => { if (!dragging) window.desk.send("feeder:mouse", true); });
  if (item.kind !== "note") return;
  const timer = setTimeout(() => void expire(paper, item.note), view?.autoTrashMs ?? 30 * 60_000);
  paper.addEventListener("dblclick", () => { clearTimeout(timer); void discard(paper); });
  enableDrag(paper, item.note, () => clearTimeout(timer));
}

function enableDrag(paper: HTMLElement, note: Note, onDetach: () => void): void {
  paper.addEventListener("pointerdown", (e) => {
    if (e.button !== 0 || (e.target as HTMLElement).closest("button")) return;
    const sx = e.clientX, sy = e.clientY;
    const r0 = paper.getBoundingClientRect();
    let moved = false;
    paper.setPointerCapture(e.pointerId);
    dragging = true;
    const move = (ev: PointerEvent) => {
      const dx = ev.clientX - sx, dy = ev.clientY - sy;
      if (!moved && Math.hypot(dx, dy) < 6) return;
      moved = true;
      paper.classList.add("dragging", "loose");
      paper.style.transform = `translate(${dx}px, ${dy}px) rotate(${Math.max(-6, Math.min(6, dx / 40))}deg)`;
    };
    const up = (ev: PointerEvent) => {
      paper.removeEventListener("pointermove", move);
      paper.removeEventListener("pointerup", up);
      dragging = false;
      const dx = ev.clientX - sx, dy = ev.clientY - sy;
      if (!moved || Math.hypot(dx, dy) < 40) {
        paper.style.transform = "";
        paper.classList.remove("dragging", "loose");
        return;
      }
      onDetach();
      window.desk.send("feeder:detach", {
        note,
        x: Math.round(window.screenX + r0.left + dx),
        y: Math.round(window.screenY + r0.top + dy),
        w: Math.ceil(r0.width) + 16,
        h: Math.ceil(r0.height) + 16,
      });
      remove(paper);
    };
    paper.addEventListener("pointermove", move);
    paper.addEventListener("pointerup", up);
  });
}

function remove(paper: HTMLElement): void {
  hung.delete(paper);
  paper.remove();
  window.desk.send("feeder:mouse", true);
  maybeEmpty();
}

function maybeEmpty(): void {
  if (!hung.size && !printing && !queue.length) window.desk.send("feeder:empty");
}

  sfx.crumple();
  paper.classList.add("crumple");
  await wait(450);
  remove(paper);
}

async function expire(paper: HTMLElement, note: Note): Promise<void> {

  paper.style.transformOrigin = "50% 50%";
  paper.classList.add("crumple");
  await wait(450);
  window.desk.send("feeder:expired", {
    note,
    x: Math.round(window.screenX + r.left + r.width / 2),
    y: Math.round(window.screenY + r.top + r.height / 2),
  });
  remove(paper);
}

void refresh();
```

```ts
  let x = wx, y = wy, raf = 0;
  paper.setPointerCapture(e.pointerId);
  const move = (ev: PointerEvent) => {
    x = wx + ev.screenX - sx;
    y = wy + ev.screenY - sy;
    if (!raf) raf = requestAnimationFrame(() => { raf = 0; window.desk.send("note:move", { id, x, y }); });
  };
  const up = () => {
    paper.removeEventListener("pointermove", move);
    paper.removeEventListener("pointerup", up);
    if (x !== wx || y !== wy) window.desk.send("note:moved", { id, x, y });
  };
  paper.addEventListener("pointermove", move);
  paper.addEventListener("pointerup", up);
});

paper.addEventListener("dblclick", async () => {
  sfx.crumple();
  paper.style.transformOrigin = "50% 50%";
  paper.classList.add("crumple");
```

```ts
import type { AppView, FlyRequest } from "../../shared/ipc-types";
import { setVolume, sfx } from "../sound";

void window.desk.invoke<AppView>("state:get").then((v) => setVolume(v.settings.volume));

window.desk.on("fx:fly", (r: FlyRequest) => {
  const ball = document.createElement("div");
  ball.className = "ball";
  document.body.append(ball);
  const fx = r.from.x - r.origin.x, fy = r.from.y - r.origin.y;
  const tx = r.to.x - r.origin.x, ty = r.to.y - r.origin.y;
  const peak = Math.min(fy, ty) - 220;
  const frames: Keyframe[] = [];
  for (let i = 0; i <= 24; i++) {
    const t = i / 24;
    const x = fx + (tx - fx) * t;
    const y = (1 - t) ** 2 * fy + 2 * (1 - t) * t * peak + t * t * ty; // quadratic arc
    frames.push({ transform: `translate(${x - 28}px, ${y - 28}px) rotate(${t * 540}deg) scale(${1 - 0.55 * t})` });
  }
  const anim = ball.animate(frames, { duration: 950, easing: "cubic-bezier(.45,.05,.55,.95)", fill: "forwards" });
  anim.onfinish = () => {
    sfx.drop();
    ball.remove();
    window.desk.send("fx:done", r.id);
  };
});
```

```ts
import type { AppView, Settings } from "../../shared/ipc-types";

const $ = <T extends HTMLElement>(id: string) => document.getElementById(id) as T;
let view: AppView;

function el<K extends keyof HTMLElementTagNameMap>(tag: K, props: Partial<HTMLElementTagNameMap[K]> = {}, ...kids: (Node | string)[]) {
  const e = Object.assign(document.createElement(tag), props);
  e.append(...kids);
  return e;
}

const REASONS: Record<string, string> = {
  bad_code: "配对码格式不对", code_not_found: "没有这个配对码", self: "这是你自己的码",

  nickname_required: "名字不能为空",
};
```
